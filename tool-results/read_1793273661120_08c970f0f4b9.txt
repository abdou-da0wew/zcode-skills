   665→  const nebChars = "·∙∶⋮⋯░▒";
   666→  function render(){
   667→    t += 0.012;
   668→    const bufNeb = newBuf();
   669→    const bufStar = newBuf();
   670→    const bufPlanet = newBuf();
   671→    const bufAst = newBuf();
   672→    const bufComet = newBuf();
   673→
   674→    // 星云气体
   675→    for(const n of nebulae){
   676→      const cx = n.x + Math.sin(t*0.2 + n.seed)*3;
   677→      const cy = n.y + Math.cos(t*0.15 + n.seed)*2;
   678→      for(let dy=-n.r; dy<=n.r; dy++){
   679→        for(let dx=-n.r*1.6; dx<=n.r*1.6; dx++){
   680→          const xx = Math.floor(cx+dx);
   681→          const yy = Math.floor(cy+dy);
   682→          if(xx<0||xx>=cols||yy<0||yy>=rows) continue;
   683→          const d = Math.sqrt((dx/1.6)*(dx/1.6) + dy*dy);
   684→          if(d > n.r) continue;
   685→          const noise = Math.sin(xx*0.3 + n.seed)*Math.cos(yy*0.4 - n.seed*0.7) + Math.sin(t*0.5+n.seed);
   686→          const intensity = (1 - d/n.r) * (0.5 + 0.5*noise);
   687→          if(intensity > 0.55){
   688→            const ci = Math.min(nebChars.length-1, Math.floor(intensity * nebChars.length));
   689→            bufNeb[yy][xx] = nebChars[ci];
   690→          }
   691→        }
   692→      }
   693→    }
   694→
   695→    // 闪烁星点
   696→    for(const s of stars){
   697→      const xi = Math.floor((s.x + Math.sin(t*0.3 + s.y)*0.3 + cols)%cols);
   698→      const yi = Math.floor(s.y);
   699→      if(yi>=0 && yi<rows){
   700→        const tw = (Math.sin(t*2 + s.x*0.7 + s.y*0.3)+1)*0.5;
   701→        if(tw>0.55 || s.c==='★' || s.c==='✦') bufStar[yi][xi] = s.c;
   702→      }
   703→    }
   704→
   705→    // 旋转行星
   706→    const cx = cols/2, cy = rows/2;
   707→    const R = Math.min(cols,rows*2)*0.18;
   708→    const aspect = 0.5;
   709→    for(let phi=0; phi<Math.PI; phi+=0.06){
   710→      for(let theta=0; theta<2*Math.PI; theta+=0.04){
   711→        const x3 = R*Math.sin(phi)*Math.cos(theta + t);
   712→        const y3 = R*Math.cos(phi);
   713→        const z3 = R*Math.sin(phi)*Math.sin(theta + t);
   714→        const lx = Math.cos(t*0.6), ly = -0.3, lz = Math.sin(t*0.6);
   715→        const nx = Math.sin(phi)*Math.cos(theta + t);
   716→        const ny = Math.cos(phi);
   717→        const nz = Math.sin(phi)*Math.sin(theta + t);
   718→        const lum = nx*lx + ny*ly + nz*lz;
   719→        if(z3 < 0) continue;
   720→        const sx = Math.floor(cx + x3/aspect);
   721→        const sy = Math.floor(cy + y3);
   722→        if(sx<0||sx>=cols||sy<0||sy>=rows) continue;
   723→        const idx = Math.max(0, Math.min(chars.length-1, Math.floor((lum+0.2)*chars.length*0.9)));
   724→        bufPlanet[sy][sx] = chars[idx];
   725→      }
   726→    }
   727→
   728→    // 行星轨道与卫星
   729→    for(let i=0; i<3; i++){
   730→      const orbR = R*(1.6 + i*0.5);
   731→      const tilt = 0.35 + i*0.1;
   732→      for(let a=0; a<2*Math.PI; a+=0.04){
   733→        const ox = orbR*Math.cos(a + t*(0.4-i*0.1));
   734→        const oy = orbR*Math.sin(a + t*(0.4-i*0.1))*tilt;
   735→        const sx = Math.floor(cx + ox/aspect);
   736→        const sy = Math.floor(cy + oy);
   737→        if(sx<0||sx>=cols||sy<0||sy>=rows) continue;
   738→        if(bufPlanet[sy][sx]===' ') bufPlanet[sy][sx] = (i===0?'.':(i===1?'∙':'•'));
   739→      }
   740→      const a = t*(0.7-i*0.18);
   741→      const ox = orbR*Math.cos(a);
   742→      const oy = orbR*Math.sin(a)*tilt;
   743→      const sx = Math.floor(cx + ox/aspect);
   744→      const sy = Math.floor(cy + oy);
   745→      if(sx>=0&&sx<cols&&sy>=0&&sy<rows) bufPlanet[sy][sx] = (i===0?'◉':(i===1?'◎':'○'));
   746→    }
   747→
   748→    // 彗星拖尾
   749→    for(const c of comets){
   750→      c.x += c.vx; c.y += c.vy; c.life++;
   751→      if(c.x>cols+10 || c.y>rows+5){
   752→        c.x = -10; c.y = Math.random()*rows*0.5;
   753→        c.life = 0;
   754→      }
   755→      const tail = 16;
   756→      for(let k=0;k<tail;k++){
   757→        const tx = Math.floor(c.x - k*c.vx*1.5);
   758→        const ty = Math.floor(c.y - k*c.vy*1.5);
   759→        if(tx<0||tx>=cols||ty<0||ty>=rows) continue;
   760→        const tch = k===0?'☄':(k<3?'*':(k<7?'∙':(k<12?'·':'.')));
   761→        bufComet[ty][tx] = tch;
   762→      }
   763→    }
   764→
   765→    // 漂浮的小行星
   766→    for(const a of asteroids){
   767→      a.x += a.vx; a.y += a.vy; a.rot += a.rs;
   768→      if(a.x<-6) a.x = cols+6;
   769→      if(a.x>cols+6) a.x = -6;
   770→      if(a.y<-4) a.y = rows+4;
   771→      if(a.y>rows+4) a.y = -4;
   772→      const wob = Math.sin(t*0.8 + a.rot)*0.6;
   773→      drawAsteroid(bufAst, a.x + wob, a.y + Math.cos(t*0.6+a.rot)*0.4, a.shape);
   774→    }
   775→
   776→    elNeb.textContent = flush(bufNeb);
   777→    elStar.textContent = flush(bufStar);
   778→    elPlanet.textContent = flush(bufPlanet);
   779→    elAst.textContent = flush(bufAst);
   780→    elComet.textContent = flush(bufComet);
   781→    requestAnimationFrame(render);
   782→  }
   783→  render();
   784→})();

   853→/* ============================================================
   854→   3. PROTEXT ：文本乱码-解码效果
   855→   悬停或进入视口时，文字先打乱再逐字"解密"显现
   856→   ============================================================ */
   857→(function(){
   858→  const glyphs = "★☆✶✦✧◉◎○●∙•~≈≋▁▂▃▄▅▆▇█ABCDEFGHJKLMNPQRSTUVWXYZ0123456789@#%&*";
   859→  const RAND = ()=> glyphs[Math.floor(Math.random()*glyphs.length)];
   860→
   861→  function scramble(el){
   862→    if(el.dataset.scrLock==='1') return;
   863→    el.dataset.scrLock='1';
   864→    el.classList.add('scrambling');
   865→    const target = el.dataset.text || el.textContent;
   866→    const chars = [...target];
   867→    const len = chars.length;
   868→    let frame = 0;
   869→    const total = 16;
   870→    const step = ()=>{
   871→      frame++;
   872→      const out = chars.map((ch,i)=>{
   873→        if(ch===' '||ch==='\n') return ch;
   874→        const reveal = Math.floor((frame/total)*len*1.2);
   875→        if(i < reveal) return ch;
   876→        return RAND();
   877→      }).join('');
   878→      el.textContent = out;
   879→      if(frame < total){
   880→        setTimeout(step, 40);
   881→      } else {
   882→        el.textContent = target;
   883→        el.classList.remove('scrambling');
   884→        el.dataset.scrLock='0';
   885→      }
   886→    };
   887→    step();
   888→  }
   889→
   890→  document.querySelectorAll('.pt').forEach(el=>{
   891→    if(!el.dataset.text) el.dataset.text = el.textContent.trim();
   892→    el.addEventListener('mouseenter', ()=>scramble(el));
   893→  });
   894→
   895→  const io = new IntersectionObserver((entries)=>{
   896→    entries.forEach(e=>{
   897→      if(e.isIntersecting){
   898→        scramble(e.target);
   899→        io.unobserve(e.target);
   900→      }
   901→    });
   902→  },{threshold:0.3});
   903→  document.querySelectorAll('.intro-text .pt, .spec-row .pt, .showcase-text .pt').forEach(el=>io.observe(el));
   904→})();
