   580→  return parts.length ? parts[parts.length - 1] : '未分类';

   619→    if (!activeCats.has(cat)) return;

   621→    const kp = getKpLeaf(q);              // 三级：知识点叶子
   622→    if (!groups[cat]) groups[cat] = {};
   623→    if (!groups[cat][law]) groups[cat][law] = {};
   624→    if (!groups[cat][law][kp]) groups[cat][law][kp] = [];
   625→    groups[cat][law][kp].push(q);
   626→  });
   627→
   628→  const container = document.getElementById('leftScroll');
   629→  container.innerHTML = '';
   630→  Object.keys(groups).forEach(cat => {
   631→    const laws = groups[cat];
   632→    const lawKeys = Object.keys(laws);
   633→    const total = Object.values(laws).reduce(
   634→      (s, kps) => s + Object.values(kps).reduce((n, arr) => n + arr.length, 0), 0
   635→    );
   636→
   637→    // 兜底：如果该一级下没有任何有效二级 / 三级（法规名都是"通用" 且 知识点都是"未分类"），
   638→    // 直接平铺题号，避免出现一个孤零零的"未分类"中间层
   639→    const hasLaw = lawKeys.some(l => l && l !== '通用');
   640→    const hasKp = lawKeys.some(l => Object.keys(laws[l]).some(k => k && k !== '未分类'));
   641→    const flat = !hasLaw && !hasKp;
   642→
   643→    let html = `<div class="cat-group"><div class="cat-top">📂 ${escapeHtml(cat)} <span class="cat-count">${total}题</span></div>`;
   644→
   645→    if (flat) {
   646→      // 收集所有题号，按 id 排序后一把铺出来
   647→      const allQs = [];
   648→      lawKeys.forEach(l => Object.values(laws[l]).forEach(arr => allQs.push(...arr)));
   649→      allQs.sort((a, b) => a.id - b.id);
   650→      html += `<div class="cat-sub cat-sub-flat"><div class="cat-nums">`;
   651→      allQs.forEach(q => {
   652→        const st = states[q.id] || 'unstudied';
   653→        html += `<span class="qnum ${st}" data-id="${q.id}" onclick="selectQ(${q.id})">${q.id}</span>`;
   654→      });
   655→      html += `</div></div>`;
   656→    } else {
   657→      lawKeys.forEach(law => {
   658→        const kps = laws[law];
   659→        const lawTotal = Object.values(kps).reduce((n, arr) => n + arr.length, 0);
   660→        // 法规名长，可能换行；用 title 兜底显示完整原文
   661→        if (law && law !== '通用') {
   662→          html += `<div class="cat-law" title="${escapeHtml(law)}"><span class="cat-law-name">${escapeHtml(law)}</span><span class="cat-law-count">${lawTotal}</span></div>`;
   663→        }
   664→        Object.keys(kps).forEach(kp => {
   665→          html += `<div class="cat-sub"><div class="cat-sub-name" title="${escapeHtml(kp)}">${escapeHtml(kp)}</div><div class="cat-nums">`;
   666→          kps[kp].forEach(q => {
   667→            const st = states[q.id] || 'unstudied';
   668→            html += `<span class="qnum ${st}" data-id="${q.id}" onclick="selectQ(${q.id})">${q.id}</span>`;
   669→          });
   670→          html += `</div></div>`;
   671→        });
   672→      });
   673→    }
   674→
   675→    html += `</div>`;
   676→    container.insertAdjacentHTML('beforeend', html);
   677→  });
   678→}

   841→window.submitChoice = function (id) {
   842→  const q = QS.find(x => x.id === id);
   843→  if (!q) return;
   844→  const key = window.__pendingChoice;
   845→  if (!key) { showRetryHint('先选一个选项呀～ 😊'); return; }
   846→  const stdU = String(q.answer || '').toUpperCase();
   847→  const correct = key.toUpperCase() === stdU;
   848→
   849→  if (correct) {
   850→    answered[id] = key;
   851→    window.__wrongTriedOnce[id] = false;
   852→    showChoiceAnswer(q, key, true);
   853→    autoUpdateState(q, true);
   854→  } else {
   855→    if (!window.__wrongTriedOnce[id]) {
   856→      // 第一次错：给一次重试机会，不锁，不标错题
   857→      window.__wrongTriedOnce[id] = true;
   858→      const errOpt = document.querySelector(`#qOpts .opt[data-key="${key}"]`);
   859→      if (errOpt) {
   860→        errOpt.classList.add('wrong-shake');
   861→        setTimeout(() => errOpt.classList.remove('wrong-shake'), 600);
   862→      }
   863→      showRetryHint('🤔 不对哦，再想想？（这次错不算，可以重选）');
   864→      window.__pendingChoice = null;
   865→      if (errOpt) errOpt.classList.remove('selected');
   866→      return;
   867→    }
   868→    // 第二次错：真正锁定 + 标错题
   869→    answered[id] = key;
   870→    showChoiceAnswer(q, key, false);
   871→    autoUpdateState(q, false);
   872→  }

   945→function fuzzyMatch(a, b) {
   946→  if (a === null || b === null || a === undefined || b === undefined) return false;
   947→  const norm = s => String(s).replace(/[\s\.,，。、；;:：!！?？""''""()（）]/g, '').toLowerCase();
   948→  return norm(a) === norm(b);
   949→}

  1012→  document.getElementById('qStateBtn').textContent = STATE_LABELS[next];
  1013→  applyStatesToDOM();
  1014→  applyFilters();
  1015→};

  1342→/* ===== 工具 ===== */
  1343→function escapeHtml(s) {
  1344→  if (s === null || s === undefined) return '';
  1345→  return String(s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
  1346→}
  1347→function renderMD(s) {
  1348→  if (s === null || s === undefined) return '';
  1349→  return escapeHtml(s)
  1350→    .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
  1351→    .replace(/\*(.+?)\*/g, '<em>$1</em>')
  1352→    .replace(/`(.+?)`/g, '<code>$1</code>')
  1353→    .replace(/\n/g, '<br>')
  1354→    .replace(/____+/g, '<span style="display:inline-block;min-width:80px;border-bottom:2px solid currentColor;margin:0 4px;opacity:.5;">&nbsp;</span>');
  1355→}
