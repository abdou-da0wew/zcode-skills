    77→            breakdown[name] = breakdown.get(name, 0) + price

    90→def _request(token: str, method: str, path: str,
    91→             params: Optional[dict] = None,
    92→             body: Optional[dict] = None) -> Any:
    93→    """Send an HTTP request and return the parsed JSON data (with retries)."""
    94→    url = BASE_URL + path
    95→    headers = {
    96→        "Authorization": token,
    97→        "X-Platform": "openclaw",
    98→        "Content-Type": "application/json;charset=utf-8",
    99→    }
   100→
   101→    if method.upper() == "GET" and params:
   102→        query = urllib.parse.urlencode(
   103→            {k: (json.dumps(v) if isinstance(v, (list, dict)) else v)
   104→             for k, v in params.items() if v is not None}
   105→        )
   106→        url = f"{url}?{query}"
   107→
   108→    data = json.dumps(body).encode("utf-8") if body else None
   109→    req = urllib.request.Request(url, data=data, headers=headers, method=method.upper())
   110→
   111→    last_error_result = None
   112→    for attempt in range(1, MAX_RETRIES + 1):
   113→        try:
   114→            with urllib.request.urlopen(req, timeout=REQUEST_TIMEOUT_SECONDS) as resp:
   115→                raw = resp.read().decode("utf-8")
   116→                return json.loads(raw)
   117→        except urllib.error.HTTPError as e:
   118→            body_bytes = e.read()
   119→            try:
   120→                err = json.loads(body_bytes)
   121→            except Exception:
   122→                err = body_bytes.decode("utf-8", errors="replace")
   123→            retryable = e.code in RETRYABLE_HTTP_STATUS
   124→            print(f"[HTTP {e.code}] {e.reason}: {err}", file=sys.stderr)
   125→            last_error_result = {
   126→                "code": e.code, "success": False,
   127→                "msg": str(e.reason), "error": err, "retryable": retryable,
   128→            }
   129→            if not retryable:
   130→                return last_error_result
   131→        except urllib.error.URLError as e:
   132→            reason = str(getattr(e, "reason", e))
   133→            print(f"[Request failed] {reason}", file=sys.stderr)
   134→            last_error_result = {
   135→                "code": -1, "success": False,
   136→                "msg": "network_error", "error": reason, "retryable": True,
   137→            }
   138→        except TimeoutError as e:
   139→            print(f"[Request timeout] {e}", file=sys.stderr)
   140→            last_error_result = {
   141→                "code": -1, "success": False,
   142→                "msg": "timeout", "error": str(e), "retryable": True,
   143→            }
   144→        except Exception as e:
   145→            print(f"[Request failed] {e}", file=sys.stderr)
   146→            return {
   147→                "code": -1, "success": False,
   148→                "msg": "unknown_error", "error": str(e), "retryable": False,
   149→            }
   150→
   151→        if attempt < MAX_RETRIES:
   152→            backoff = (2 ** (attempt - 1)) + random.uniform(0, 0.3)
   153→            print(f"[Retry] attempt={attempt}/{MAX_RETRIES} wait={backoff:.2f}s", file=sys.stderr)
   154→            time.sleep(backoff)
   155→
   156→    return last_error_result or {
   157→        "code": -1, "success": False,
   158→        "msg": "request_failed", "error": "max retries exceeded", "retryable": True,
   159→    }

   163→    """Pretty-print JSON result."""
   164→    print(json.dumps(data, ensure_ascii=False, indent=2))

   480→# Combined Workflows
   481→# ──────────────────────────────────────────────────────────────────────────────
   482→
   483→def workflow_scholar_profile(token: str, name: str) -> dict:
   484→    """
   485→    Workflow 1: Scholar Profile
   486→    Search scholar → details + portrait + papers + patents + projects
   487→    """
   488→    print(f"[1/6] Searching scholar: {name}", file=sys.stderr)
   489→    search_result = person_search(token, name=name, size=5)
   490→    if not search_result or not search_result.get("data"):
   491→        return {"error": f"Scholar not found: {name}"}
   492→
   493→    candidates = search_result["data"]
   494→    scholar = candidates[0]
   495→    person_id = scholar.get("id") or scholar.get("_id")
   496→    print(f"      Found: {scholar.get('name')} ({scholar.get('org')}), ID={person_id}", file=sys.stderr)

   553→def workflow_paper_deep_dive(token: str, title: str = None, keyword: str = None,
   554→                              author: str = None, order: str = "n_citation") -> dict:
   555→    """
   556→    Workflow 2: Paper Deep Dive
   557→    Search paper → details + citation chain + basic info of cited papers
   558→    """
   559→    print(f"[1/4] Searching paper: title={title}, keyword={keyword}", file=sys.stderr)
   560→    if keyword or author:
   561→        search_result = paper_search_pro(token, title=title, keyword=keyword,
   562→                                         author=author, order=order, size=5)
   563→        search_api = "paper_search_pro"
   564→    else:
   565→        search_result = paper_search(token, title=title or keyword, size=5)
   566→        search_api = "paper_search"
   567→        if not search_result or not search_result.get("data"):
   568→            # Fall back to pro search when title search yields no results to improve recall
   569→            print("      Title search returned no results; falling back to paper_search_pro...", file=sys.stderr)
   570→            search_result = paper_search_pro(token, title=title, keyword=title,
   571→                                             author=author, order=order, size=5)
   572→            search_api = "paper_search_pro(fallback)"
   573→
   574→    if not search_result or not search_result.get("data"):
   575→        return {"error": "No relevant papers found"}
   576→
   577→    papers = search_result["data"]
   578→    top_paper = papers[0]
   579→    paper_id = top_paper.get("id") or top_paper.get("_id")
   580→    print(f"      Found: {top_paper.get('title')[:60]}, ID={paper_id}", file=sys.stderr)
   581→
   582→    result = {
   583→        "source_api_chain": [
   584→            search_api,
   585→            "paper_detail",
   586→            "paper_relation",
   587→            "paper_info",
   588→        ],
   589→        "search_candidates": papers[:5],
   590→        "selected_id": paper_id,
   591→        "selected_title": top_paper.get("title"),
   592→    }
   593→
   594→    print("[2/4] Fetching paper details...", file=sys.stderr)
   595→    detail = paper_detail(token, paper_id)
   596→    if detail and detail.get("data"):
   597→        result["detail"] = detail["data"]
   598→
   599→    print("[3/4] Fetching citation relationships...", file=sys.stderr)
   600→    relation = paper_relation(token, paper_id)
   601→    if relation and relation.get("data"):
   602→        # data structure: [{"_id": "<paper_id>", "cited": [{...}, ...]}]
   603→        # the outer array wraps each paper; the actual citation list is in the cited field
   604→        all_cited = []
   605→        for item in relation["data"]:
   606→            all_cited.extend(item.get("cited") or [])
   607→        result["citations_count"] = len(all_cited)
   608→        result["citations_preview"] = all_cited[:10]
   609→
   610→        cited_ids = [c.get("_id") or c.get("id") for c in all_cited[:20]
   611→                     if c.get("_id") or c.get("id")]
   612→        if cited_ids:
   613→            print(f"[4/4] Batch-fetching basic info for {len(cited_ids)} cited papers...", file=sys.stderr)
   614→            info = paper_info(token, cited_ids)
   615→            if info and info.get("data"):
   616→                result["cited_papers_info"] = info["data"]
   617→        else:
   618→            print("[4/4] Skipping (no cited IDs)", file=sys.stderr)
   619→    else:
   620→        print("[4/4] Skipping (no citation data)", file=sys.stderr)
   621→
   622→    return result

   625→def workflow_org_analysis(token: str, org: str) -> dict:
   626→    """
   627→    Workflow 3: Org Analysis
   628→    Org disambiguation pro → details + scholars + papers + patents
   629→    """
   630→    print(f"[1/5] Disambiguating org: {org}", file=sys.stderr)
   631→    disamb = org_disambiguate_pro(token, org)
   632→    org_id = None
   633→
   634→    if disamb and disamb.get("data"):
   635→        data = disamb["data"]
   636→        if isinstance(data, list) and data:
   637→            first = data[0]
   638→            org_id = first.get("一级ID") or first.get("二级ID")
   639→        elif isinstance(data, dict):
   640→            org_id = data.get("一级ID") or data.get("二级ID")
   641→
   642→    if not org_id:
   643→        print("      Disambiguation pro returned no ID; trying org search...", file=sys.stderr)
   644→        search_r = org_search(token, [org])
   645→        if search_r and search_r.get("data"):
   646→            orgs = search_r["data"]
   647→            org_id = orgs[0].get("org_id") if orgs else None
   648→
   649→    if not org_id:
   650→        return {"error": f"Could not find org ID: {org}"}
   651→
   652→    print(f"      Org ID: {org_id}", file=sys.stderr)
   653→    result = {
   654→        "source_api_chain": [
   655→            "org_disambiguate_pro",
   656→            "org_detail",
   657→            "org_person_relation",
   658→            "org_paper_relation",
   659→            "org_patent_relation",
   660→        ],
   661→        "org_query": org,
   662→        "org_id": org_id,
   663→        "disambiguate": disamb,
   664→    }
   665→
   666→    print("[2/5] Fetching org data (parallel)...", file=sys.stderr)
   667→    tasks = {
   668→        "detail": lambda: org_detail(token, [org_id]),
   669→        "scholars": lambda: org_person_relation(token, org_id, offset=0),
   670→        "papers": lambda: org_paper_relation(token, org_id, offset=0),
   671→        "patents": lambda: org_patent_relation(token, org_id, page=1, page_size=100),
   672→    }
   673→    with ThreadPoolExecutor(max_workers=4) as pool:
   674→        futures = {pool.submit(fn): key for key, fn in tasks.items()}
   675→        for future in as_completed(futures):
   676→            key = futures[future]
   677→            try:
   678→                resp = future.result()
   679→            except Exception as e:
   680→                print(f"  [{key}] failed: {e}", file=sys.stderr)
   681→                continue
   682→            data = resp.get("data") if resp else None
   683→            if not data:
   684→                continue
   685→            if key == "detail":
   686→                result["detail"] = data
   687→            elif key == "scholars":
   688→                result["scholars"] = data
   689→                result["scholars_total"] = resp.get("total", len(data))
   690→            elif key == "papers":
   691→                result["papers"] = data
   692→                result["papers_total"] = resp.get("total", len(data))
   693→            elif key == "patents":
   694→                result["patents"] = data
   695→                result["patents_total"] = resp.get("total", len(data))
   696→
   697→    return result

   739→def workflow_paper_qa(token: str, query: str = None,
   740→                      topic_high: str = None, topic_middle: str = None,
   741→                      sci_flag: bool = False, sort_citation: bool = False, sort_year: bool = False,
   742→                      author_id: list = None, org_id: list = None, venue_ids: list = None,
   743→                      size: int = 10,
   744→                      year_from: int = None, year_to: int = None,
   745→                      min_citations: int = None, cursor: str = None) -> dict:
   746→    """
   747→    Workflow: Paper QA Search — prefer Pro, use legacy sparingly.
   748→    - Default: natural language / filters → paper_qa_search_pro (¥0.70, fixed 10/page)
   749→    - Rare: only topic_high/topic_middle → legacy paper_qa_search (¥0.05)
   750→    """
   751→    # Structured topic_* exists only on the legacy endpoint; avoid unless required.
   752→    if topic_high or topic_middle:
   753→        print(f"[1/1] Legacy topic QA search: topic_high={bool(topic_high)}", file=sys.stderr)
   754→        qa_result = paper_qa_search(
   755→            token, query=query, use_topic=True,
   756→            topic_high=topic_high, topic_middle=topic_middle,
   757→            sci_flag=sci_flag, force_citation_sort=sort_citation,
   758→            force_year_sort=sort_year,
   759→            author_id=author_id, org_id=org_id, venue_ids=venue_ids,
   760→            size=size,
   761→        )
   762→        if isinstance(qa_result, dict):
   763→            qa_result["source_api_chain"] = ["paper_qa_search"]
   764→            qa_result["route"] = "paper_qa_search"
   765→        return qa_result
   766→
   767→    sort = None
   768→    if sort_citation:
   769→        sort = "citation"
   770→    elif sort_year:
   771→        sort = "recent"
   772→
   773→    print(f"[1/1] Paper QA Search Pro: query={query}, sort={sort}", file=sys.stderr)
   774→    qa_result = paper_qa_search_pro(
   775→        token,
   776→        query=query,
   777→        query_type="auto",
   778→        cursor=cursor,
   779→        author_ids=author_id,
   780→        organization_ids=org_id,
   781→        venue_ids=venue_ids,
   782→        year_from=year_from,
   783→        year_to=year_to,
   784→        min_citations=min_citations,
   785→        sort=sort,
   786→    )
   787→    data = (qa_result or {}).get("data") or {}
   788→    items = data.get("items") if isinstance(data, dict) else None
   789→    if qa_result and qa_result.get("code") == 200 and items:
   790→        qa_result["source_api_chain"] = ["paper_qa_search_pro"]
   791→        qa_result["route"] = "paper_qa_search_pro"
   792→        return qa_result
   793→
   794→    if query and not cursor:
   795→        print("      paper_qa_search_pro returned no results; falling back to paper_search_pro...", file=sys.stderr)
   796→        order = "n_citation" if sort_citation or not sort_year else "year"
   797→        fallback = paper_search_pro(token, keyword=query, order=order, size=min(size, 10))
   798→        fb_data = (fallback or {}).get("data") or []
   799→        return {
   800→            "code": 200 if fb_data else (qa_result or {}).get("code", -1),
   801→            "success": bool(fb_data),
   802→            "msg": "" if fb_data else "no data",
   803→            "data": fb_data,
   804→            "total": (fallback or {}).get("total", len(fb_data)),
   805→            "route": "paper_qa_search_pro -> paper_search_pro",
   806→            "source_api_chain": ["paper_qa_search_pro", "paper_search_pro"],
   807→            "primary_result": qa_result,
   808→        }
   809→
   810→    if isinstance(qa_result, dict):
   811→        qa_result["source_api_chain"] = ["paper_qa_search_pro"]
   812→        qa_result["route"] = "paper_qa_search_pro"
   813→    return qa_result

   816→def workflow_patent_search(token: str, query: str, page: int = 0, size: int = 10) -> dict:
   817→    """
   818→    Workflow 6: Patent Search and Details
   819→    Patent search → retrieve details for each patent
   820→    """
   821→    print(f"[1/2] Searching patents: {query}", file=sys.stderr)
   822→    search_result = patent_search(token, query, page=page, size=size)
   823→    if not search_result or not search_result.get("data"):
   824→        return {"error": f"No patents found: {query}"}
   825→
   826→    patents = search_result["data"]
   827→    result = {
   828→        "source_api_chain": ["patent_search", "patent_detail"],
   829→        "search_results": patents,
   830→        "total": len(patents),
   831→    }
   832→
   833→    top_patents = [p for p in patents[:3] if p.get("id")]
   834→    print(f"[2/2] Fetching details for {len(top_patents)} patents (parallel)...", file=sys.stderr)
   835→    details = []
   836→    with ThreadPoolExecutor(max_workers=3) as pool:
   837→        futs = {pool.submit(patent_detail, token, p["id"]): p["id"] for p in top_patents}
   838→        for fut in as_completed(futs):
   839→            try:
   840→                d = fut.result()
   841→                if d and d.get("data"):
   842→                    details.append(d["data"])
   843→            except Exception as e:
   844→                print(f"  [patent_detail] failed: {e}", file=sys.stderr)
   845→    result["details"] = details
   846→    return result
